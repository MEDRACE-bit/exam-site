-- إضافة عملات لطالب برقم تليفونه (للأدمن بس) - بتتسجّل في wallet_transactions -- رقم سالب = خصم (ومش بينزل الرصيد تحت صفر) create or replace function public.admin_add_coins(p_phone text, p_amount int) returns json language plpgsql security definer set search_path = public as $$ declare p profiles%rowtype; begin if not exists (select 1 from profiles where id = auth.uid() and is_admin) then raise exception 'not allowed'; end if; if p_amount is null or p_amount = 0 or abs(p_amount) > 100000 then return json_build_object('ok', false, 'reason', 'bad_amount'); end if; select * into p from profiles where phone = trim(p_phone) limit 1; if not found then return json_build_object('ok', false, 'reason', 'not_found'); end if; if p.balance + p_amount < 0 then return json_build_object('ok', false, 'reason', 'low_balance', 'balance', p.balance); end if; update profiles set balance = balance + p_amount where id = p.id; insert into wallet_transactions (user_id, amount, kind, note) values (p.id, p_amount, 'topup', 'admin'); return json_build_object('ok', true, 'name', p.name, 'balance', p.balance + p_amount); end $$; revoke execute on function public.admin_add_coins(text, int) from public, anon; grant execute on function public.admin_add_coins(text, int) to authenticated;
